\begin{theorem}
$((P_1 \OR P_2) \AND (P_1 \IMP Q) \AND (P_2 \IMP Q)) \IMP ((P_1 \OR P_2) \IMP Q)$
\end{theorem}

\begin{proof}
\leavevmode
\begin{enumerate}
    \item $P_1 \OR P_2$ \hfill (Assumption)
    \item $P_1 \IMP Q$ \hfill (Assumption)
    \item $P_2 \IMP Q$ \hfill (Assumption)
    
    \item $\NOT P_1 \OR Q$ \hfill (from 2, implication elimination)
    \item $\NOT P_2 \OR Q$ \hfill (from 3, implication elimination)
    
    \item $(\NOT P_1 \OR Q) \AND (\NOT P_2 \OR Q)$ \hfill (from 4,5)
    
    \item $(\NOT P_1 \AND \NOT P_2) \OR Q$ \hfill (distribution)
    
    \item $(P_1 \OR P_2) \IMP Q$ \hfill (implication form)
\end{enumerate}
\end{proof}
